📈 FARE CALCULATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Formula: Total = 20 + (km × 8) + (min × 0.5)

Example: 15km, 45min ride, 3 participants
  Total Fare:     162.50 BDT
  Initiator:      81.25 BDT (50%)
  Passenger 1:    40.63 BDT (25%)
  Passenger 2:    40.63 BDT (25%)
